// Integration audit B: black-box browser-contract probe of the built API (local only, unreachable Supabase placeholder).
import { spawn } from 'node:child_process';

const root = process.argv[2];
const ORIGIN = 'http://localhost:5173';
const wait = ms => new Promise(r => setTimeout(r, ms));

async function start(port, extra) {
  const env = { PATH: process.env.PATH, HOME: process.env.HOME, NODE_ENV: 'development', HOST: '127.0.0.1', PORT: String(port),
    SUPABASE_URL: 'http://127.0.0.1:9', SUPABASE_ANON_KEY: 'probe-placeholder', LOG_LEVEL: 'silent', UPSTREAM_TIMEOUT_MS: '500', READINESS_TIMEOUT_MS: '500', ...extra };
  const child = spawn(process.execPath, ['apps/api/dist/server.js'], { cwd: root, env, stdio: ['ignore', 'ignore', 'pipe'] });
  let err = ''; child.stderr.on('data', d => { err += d; });
  for (let i = 0; i < 50; i++) { try { await fetch(`http://127.0.0.1:${port}/health/live`); return { child, err: () => err }; } catch { await wait(100); } }
  child.kill('SIGKILL'); throw new Error(`server did not start: ${err.slice(0, 500)}`);
}
async function call(port, method, path, headers = {}, body) {
  const r = await fetch(`http://127.0.0.1:${port}${path}`, { method, headers, body });
  const text = await r.text();
  const h = Object.fromEntries([...r.headers].filter(([k]) => k.startsWith('access-control') || ['retry-after', 'content-type', 'x-request-id', 'cache-control', 'etag', 'vary'].includes(k)));
  let json; try { json = JSON.parse(text); } catch { json = text.slice(0, 120); }
  return { status: r.status, headers: h, body: json };
}
const out = [];
const record = (name, res) => { out.push({ name, ...res }); };

{
